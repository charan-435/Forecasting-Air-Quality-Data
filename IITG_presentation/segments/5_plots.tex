\begin{frame}{Experimental results --- what we compare}
  \begin{block}{Baselines vs WaveCatBoost}
    Same data splits and horizons; only the model changes.
    \begin{itemize}\setlength\itemsep{0.15em}
      \item \textbf{Seasonal naive} (lag $S{=}24$), \textbf{SARIMA}
      \item Trees on \textbf{raw} $Y$: LightGBM, CatBoost (, XGBoost)
      \item Wave $+$ trees: WaveLGB, \textbf{WaveCatBoost} (paper model)
      \item Paper also reports RNN, TCN, TFT, Transformer (Table~1)
    \end{itemize}
  \end{block}
  \begin{itemize}
    \item \textbf{Why MASE?} pollutants have different units/scales (NO$_2$ vs PM$_{2.5}$).
          MASE is \textbf{scale-free}: error relative to a seasonal naive on train,
          so we can compare methods across pollutants.
    \item Also report MAE / RMSE in original units after inverting min-max.
  \end{itemize}
\end{frame}

\begin{frame}{Error metrics (formulas)}
  {\footnotesize
  With train length $N$, horizon $h$, seasonality $S{=}24$:
  \[
    \mathrm{MASE}
    =\frac{\tfrac1h\sum_{t=1}^{h}|\widehat Y_t-Y_t|}
          {\tfrac1{N-S}\sum_{t=S+1}^{N}|Y_t-Y_{t-S}|}
  \]
  (equivalent to paper Eq.~(6) / Hyndman seasonal MASE).\\[0.4em]
  \[
    \mathrm{MAE}=\tfrac1h\sum_{t=1}^{h}|\widehat Y_t-Y_t|,
    \qquad
    \mathrm{RMSE}=\sqrt{\tfrac1h\sum_{t=1}^{h}(\widehat Y_t-Y_t)^2}.
  \]
  }
  \vspace{0.3em}
  \begin{itemize}
    \item MASE $<1$: better than seasonal naive on the train scale.
    \item Lower is better for all three.
  \end{itemize}
\end{frame}

\begin{frame}{Did we beat the paper?}
  \begin{block}{What we can say}
    \begin{itemize}
      \item \textbf{Ranking matches Table~1:} WaveCatBoost beats LGB and CatBoost
            on every overlapping 1-step cell.
      \item Our WaveCatBoost MASE is numerically smaller than the letter's WaveCB.
    \end{itemize}
  \end{block}
  \begin{alertblock}{What we should \emph{not} claim}
    ``We beat the paper'' as a scientific result --- baselines dropped too
    (e.g.\ CPCB PM$_{2.5}$ LGB: paper 3.81 $\to$ ours 0.69). Absolute levels are
    \textbf{not digit-comparable} (hyperparams, notebook details, windowing).
  \end{alertblock}
  \begin{exampleblock}{Talk line}
    Under the letter's rolling 1-step setup we reproduce the same ranking ---
    WaveCatBoost wins --- and our WaveCB MASE is lower than Table~1, but we
    compare \textbf{gaps / ranks}, not absolute digits.
  \end{exampleblock}
\end{frame}
