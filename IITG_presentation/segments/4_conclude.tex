% ---- 1. Why conformal / what it does ----
\begin{frame}{Why conformal prediction?}
  \begin{block}{Idea in one line}
    WaveCatBoost gives one number $\widehat Y_t$ (``PM$_{2.5}$ will be 20'').\\
    Conformal adds a \textbf{band} around it: ``with about 95\% chance, the truth is between $L_t$ and $U_t$.''
  \end{block}
  \begin{itemize}
    \item \textbf{Why we need it:} a single guess is risky for alerts --- we want a safe range.
    \item \textbf{How it works :} look at recent forecast mistakes; if errors were large lately, make the band wider; if small, make it narrower.
    \item \textbf{Why not a normal-based interval?} air-quality errors are irregular (spikes, quiet stretches). Conformal does \textbf{not} assume a normal curve.
    \item \textbf{What ``95\%'' means:} over many hours, about 95 out of 100 true values should fall inside the band (coverage).
  \end{itemize}
\end{frame}

% ---- 2. Full formula + term-by-term + how to use ----
\begin{frame}{Conformal formula}
  \[
    CS_t=\frac{|Y_t-\widehat Y_t|}{\widehat U_t},
    \qquad
    CQ_t=Q_{1-\alpha}\!\big(CS_{t-\kappa+1:t}\big),
    \qquad
    PI_t=\widehat Y_t\pm CQ_t\,\widehat U_t
  \]
  {\footnotesize
  \begin{itemize}\setlength\itemsep{0.25em}
    \item $Y_t$, $\widehat Y_t$: observed value and WaveCatBoost forecast.
    \item $\widehat U_t$: rolling MAE of recent residuals (local error scale).
    \item $CS_t$: nonconformity --- absolute error scaled by $\widehat U_t$.
    \item $\alpha$: miscoverage rate ($\alpha=0.05$ $\Rightarrow$ 95\% target).
    \item $\kappa$: calibration window length (we try $\{24,168,336\}$~h).
    \item $CQ_t$: $(1-\alpha)$-quantile of the last $\kappa$ scores $CS$.
    \item $L_t=\widehat Y_t-CQ_t\widehat U_t$,\ \
          $U_t=\widehat Y_t+CQ_t\widehat U_t$
          (clip $L_t\ge 0$ for concentrations).
  \end{itemize}
  }
  \vspace{0.2em}
  \begin{exampleblock}{How to use it}
    Fit on train $\to$ compute $CS$ on a held-out calibration window $\to$
    pick $\kappa$ so cal.\ coverage $\approx 95\%$ $\to$ apply $CQ\,\widehat U$
    on the test hours. Coverage $=$ fraction of test points with $Y_t\in[L_t,U_t]$.
  \end{exampleblock}
\end{frame}

\begin{frame}{Making intervals better --- CQR}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item Paper / split conformal scales one point forecast with past errors.
    \item \textbf{CQR} (extension): fit low \& high quantile models, then conformalize\\
          $s_i=\max(\hat q_{lo}-Y_i,\; Y_i-\hat q_{hi})$.
    \item Same split protocol (calibrate off the test window).
  \end{itemize}
  \vspace{0.5em}
  {\footnotesize
  \centering
  \begin{tabular}{lcc cc}
    \toprule
    & \multicolumn{2}{c}{Split conformal} & \multicolumn{2}{c}{CQR} \\
    \cmidrule(lr){2-3}\cmidrule(lr){4-5}
    Sensor & Cov. & Width & Cov. & Width \\
    \midrule
    CPCB     & 0.68 & 8.6  & \textbf{0.89} & 10.6 \\
    ID1      & 0.90 & 17.4 & \textbf{0.98} & 18.7 \\
    Shillong & 0.90 & 127  & \textbf{0.92} & \textbf{34.6} \\
    \bottomrule
  \end{tabular}
  }
  \vspace{0.6em}
  \begin{exampleblock}{Takeaway}
    CQR lifts CPCB coverage and cuts Shillong width sharply --- next slide shows CPCB bands.
  \end{exampleblock}
\end{frame}
