\begin{frame}{EMD vs Haar MODWT (WaveLGB, 1-day PM$_{2.5}$)}
  {\footnotesize
  Adaptive IMFs folded into $D_1{\ldots}D_5,S_5$ slots (Rmd §10.2).
  Same WaveLGB head as Haar / Fourier.
  }
  \vspace{0.35em}
  {\scriptsize
  \centering
  \begin{tabular}{lcc}
    \toprule
    Sensor & Haar & EMD \\
    \midrule
    CPCB PM$_{2.5}$ & \textbf{1.031} & 7.460 \\
    ID1  PM$_{2.5}$ & \textbf{0.360} & 0.616 \\
    \bottomrule
  \end{tabular}
  }
  \vspace{0.45em}
  \begin{alertblock}{Takeaway}
    Adaptive IMFs do \emph{not} beat fixed Haar under a matched head.
    Unstable IMF counts / mode mixing hurt recursive band forecasts ---
    keep EMD as a negative control.
  \end{alertblock}
\end{frame}

\begin{frame}{Shillong A0--C3 --- Lumpyngngad takeaways}
  \begin{itemize}\setlength\itemsep{0.55em}
    \item \textbf{A2 beats A0:} other pollutants' lags help PM$_{2.5}$.
    \item \textbf{Decomposed B/C does not beat A2:} Haar on top of multivariate
          lags adds no gain at Lumpyngngad.
    \item Weather (A1 / A3) is secondary to the multivariate-lag signal.
    \item Contrast with the letter path: there Haar$+$CatBoost-per-band wins on
          CPCB/ID1; here multivariate raw lags win the Shillong bake-off.
  \end{itemize}
  \vspace{0.35em}
  \begin{exampleblock}{Role in the deck}
    Extension only --- does not replace the headline WaveCatBoost model.
  \end{exampleblock}
\end{frame}
